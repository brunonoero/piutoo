input: numeropattern(numericsimple);

var: opend0(0),opend1(0),opend2(0),opend3(0),opend4(0), opend5(0);
var: highd0(0),highd1(0),highd2(0),highd3(0),highd4(0), highd5(0);
var: lowd0(0),lowd1(0),lowd2(0),lowd3(0),lowd4(0), lowd5(0);
var: closed0(0),closed1(0),closed2(0),closed3(0),closed4(0), closed5(0);
var: range1d(0);

opend0 = OpenS(0); highd0 = HighS(0); lowd0 = LowS(0); closed0 = CloseS(0);  
opend1 = OpenS(1); highd1 = HighS(1); lowd1 = LowS(1); closed1 = CloseS(1);
opend2 = OpenS(2); highd2 = HighS(2); lowd2 = LowS(2); closed2 = CloseS(2);  
opend3 = OpenS(3); highd3 = HighS(3); lowd3 = LowS(3); closed3 = CloseS(3);
opend4 = OpenS(4); highd4 = HighS(4); lowd4 = LowS(4); closed4 = CloseS(4);  
opend5 = OpenS(5); highd5 = HighS(5); lowd5 = LowS(5); closed5 = CloseS(5);

range1d = (highd1-lowd1);

Switch(numeropattern) Begin
	case 1: 		PatternDirectionalFastSa = (highd0-opend0)> ((highd1-opend1)*0.25);
	case -1:	 	PatternDirectionalFastSa = (opend0-lowd0)> ((opend1-lowd1)*0.25);
	case 2: 		PatternDirectionalFastSa = (highd0-opend0)> ((highd1-opend1)*0.5);
	case -2:	 	PatternDirectionalFastSa = (opend0-lowd0)> ((opend1-lowd1)*0.5);
	case 3: 		PatternDirectionalFastSa = (highd0-opend0)> ((highd1-opend1)*0.75);
	case -3:		PatternDirectionalFastSa = (opend0-lowd0)> ((opend1-lowd1)*0.75);
	case 4: 		PatternDirectionalFastSa = (highd0-opend0)> ((highd1-opend1)*1);
	case -4:		PatternDirectionalFastSa = (opend0-lowd0)> ((opend1-lowd1)*1);
	case 5:	 		PatternDirectionalFastSa = (highd0-opend0)> ((highd1-opend1)*1.5);
	case -5: 		PatternDirectionalFastSa = (opend0-lowd0)> ((opend1-lowd1)*1.5);
	case 6: 		PatternDirectionalFastSa = (highd0-opend0)> ((highd1-opend1)*2);
	case -6: 		PatternDirectionalFastSa = (opend0-lowd0)> ((opend1-lowd1)*2);
	case 7:	 		PatternDirectionalFastSa = (highd0-opend0)> ((highd1-opend1)*2.5);
	case -7: 		PatternDirectionalFastSa = (opend0-lowd0)> ((opend1-lowd1)*2.5); 
	case 8: 		PatternDirectionalFastSa = (highd0-opend0)> ((highd1-opend1)*3);
	case -8: 		PatternDirectionalFastSa = (opend0-lowd0)> ((opend1-lowd1)*3);

	case 9: 		PatternDirectionalFastSa = (highd0-opend0)< (highd1-opend1);
	case -9: 		PatternDirectionalFastSa = (opend0-lowd0)< (opend1-lowd1);

	case 10:	 	PatternDirectionalFastSa = closed1>closed2 and closed2>closed3 and closed3>closed4;
	case -10:		PatternDirectionalFastSa = closed1<closed2 and closed2<closed3 and closed3<closed4;
	case 11: 		PatternDirectionalFastSa = closed1>closed2 and closed2>closed3 and closed3>closed4 and closed4>closed5;
	case -11:		PatternDirectionalFastSa = closed1<closed2 and closed2<closed3 and closed3<closed4 and closed4<closed5;
	case 12:		PatternDirectionalFastSa = highd1>highd2 and lowd1>lowd2;
	case -12:		PatternDirectionalFastSa = highd1<highd2 and lowd1<lowd2;

	case 13:	 	PatternDirectionalFastSa = closed1>closed2;
	case -13:		PatternDirectionalFastSa = closed1<closed2;
	case 14:		PatternDirectionalFastSa = closed1>opend1;
	case -14:		PatternDirectionalFastSa = closed1<opend1;


	case 15: 		PatternDirectionalFastSa = closed1>(closed2+closed2*0.5*0.01);
	case -15:		PatternDirectionalFastSa = closed1<(closed2-closed2*0.5*0.01);
	case 16: 		PatternDirectionalFastSa = closed1>(closed2+closed2*1  *0.01);
	case -16:		PatternDirectionalFastSa = closed1<(closed2-closed2*1  *0.01);
	case 17: 		PatternDirectionalFastSa = closed1>(closed2+closed2*1.5*0.01);
	case -17:		PatternDirectionalFastSa = closed1<(closed2-closed2*1.5*0.01);
	case 18:		PatternDirectionalFastSa = closed1>(closed2+closed2*2  *0.01);
	case -18:		PatternDirectionalFastSa = closed1<(closed2-closed2*2  *0.01);
	case 19:		PatternDirectionalFastSa = closed1>(closed2+closed2*2.5*0.01);
	case -19:		PatternDirectionalFastSa = closed1<(closed2-closed2*2.5*0.01);
	case 20:		PatternDirectionalFastSa = closed1>(closed2+closed2*3  *0.01);
	case -20:		PatternDirectionalFastSa = closed1<(closed2-closed2*3  *0.01);


	case 21:		PatternDirectionalFastSa = highd0>highd1;
	case -21:		PatternDirectionalFastSa = lowd0<lowd1;	
	case 22:		PatternDirectionalFastSa = highd0>(highd1+highd1*0.25*0.01);
	case -22:		PatternDirectionalFastSa = lowd0<(lowd1-lowd1*0.25*0.01);	
	case 23:		PatternDirectionalFastSa = highd0>(highd1+highd1*0.5 *0.01);
	case -23:		PatternDirectionalFastSa = lowd0<(lowd1-lowd1*0.5 *0.01);	
	case 24:		PatternDirectionalFastSa = highd0>(highd1+highd1*0.75*0.01);
	case -24:		PatternDirectionalFastSa = lowd0<(lowd1-lowd1*0.75*0.01);	
	case 25:		PatternDirectionalFastSa = highd0>(highd1+highd1*1   *0.01);
	case -25:		PatternDirectionalFastSa = lowd0<(lowd1-lowd1*1   *0.01);	
	case 26:		PatternDirectionalFastSa = highd0>(highd1+highd1*1.5 *0.01);
	case -26:		PatternDirectionalFastSa = lowd0<(lowd1-lowd1*1.5   *0.01);
	
	case 27:		PatternDirectionalFastSa = lowd0>lowd1;
	case -27:		PatternDirectionalFastSa = highd0<highd1;
	
	case 28:		PatternDirectionalFastSa = lowd0>(lowd1+lowd1*0.5*0.01);	
	case -28:		PatternDirectionalFastSa = highd0<(highd1-highd1*0.5*0.01);
	case 29:		PatternDirectionalFastSa = lowd0>(lowd1+lowd1*1  *0.01);	
	case -29:		PatternDirectionalFastSa = highd0<(highd1-highd1*1  *0.01);
	case 30:		PatternDirectionalFastSa = lowd0>(lowd1+lowd1*1.5*0.01);	
	case -30:		PatternDirectionalFastSa = highd0<(highd1-highd1*1.5*0.01);
	case 31:		PatternDirectionalFastSa = lowd0>(lowd1+lowd1*2  *0.01);		
	case -31:		PatternDirectionalFastSa = highd0<(highd1-highd1*2  *0.01);
	case 32:		PatternDirectionalFastSa = lowd0>(lowd1+lowd1*2.5*0.01);	
	case -32:		PatternDirectionalFastSa = highd0<(highd1-highd1*2.5*0.01);

	case 33:		PatternDirectionalFastSa = highd1>highd5;
	case -33:		PatternDirectionalFastSa = lowd1<lowd5;	
	
	case 34:		PatternDirectionalFastSa = highd1<highd5;
	case -34:		PatternDirectionalFastSa = lowd1>lowd5;

	case 35:		PatternDirectionalFastSa = highd1> highd2 and highd1> highd3 and highd1> highd4;
	case -35:		PatternDirectionalFastSa = lowd1 < lowd2 and lowd1 < lowd3 and lowd1 < lowd4;	
	case 36:		PatternDirectionalFastSa = lowd1 > lowd2 and lowd1 > lowd3 and lowd1 > lowd4;
	case -36:		PatternDirectionalFastSa = highd1< highd2 and highd1< highd3 and highd1< highd4;

	case 37:		PatternDirectionalFastSa = closed1>closed2 and closed2>closed3 and opend0>closed1;
	case -37:		PatternDirectionalFastSa = closed1<closed2 and closed2<closed3 and opend0<closed1;

	case 38:		PatternDirectionalFastSa = (highd1-closed1)<0.20*range1d;
	case -38:		PatternDirectionalFastSa = (closed1-lowd1)<0.20*range1d;

	case 39:		PatternDirectionalFastSa = opend0>highd1;
	case -39:		PatternDirectionalFastSa = opend0<lowd1;
	
	case 40:		PatternDirectionalFastSa = opend0>(closed1+closed1*0.25*0.01);	 
	case -40:		PatternDirectionalFastSa = opend0<(closed1-closed1*0.25*0.01);
	case 41:		PatternDirectionalFastSa = opend0>(closed1+closed1*0.5 *0.01);
	case -41:		PatternDirectionalFastSa = opend0<(closed1-closed1*0.5 *0.01);
	case 42:		PatternDirectionalFastSa = opend0>(closed1+closed1*0.75*0.01);	
	case -42:		PatternDirectionalFastSa = opend0<(closed1-closed1*0.75*0.01);
	case 43:		PatternDirectionalFastSa = opend0>(closed1+closed1*1   *0.01);	
	case -43:		PatternDirectionalFastSa = opend0<(closed1-closed1*1   *0.01);

	case 44:		PatternDirectionalFastSa = lowd1>lowd2;
	case -44:		PatternDirectionalFastSa = highd1<highd2;

	case 45:		PatternDirectionalFastSa = closed1 > opend1 and  closed2 > opend2;
	case -45:		PatternDirectionalFastSa = closed1 < opend1 and  closed2 < opend2;	
	case 46:		PatternDirectionalFastSa = closed1 > opend1 and  closed2 < opend2;	
	case -46:		PatternDirectionalFastSa = closed1 < opend1 and  closed2 > opend2;


	case 47:		PatternDirectionalFastSa = C>opend0*0.99;
	case -47:		PatternDirectionalFastSa = C<opend0*1.01;	
	case 48:		PatternDirectionalFastSa = C>opend0*0.995;
	case -48:		PatternDirectionalFastSa = C<opend0*1.005;	
	case 49:		PatternDirectionalFastSa = C>opend0;
	case -49:		PatternDirectionalFastSa = C<opend0;	
	case 50:		PatternDirectionalFastSa = C>opend0*1.005;
	case -50:		PatternDirectionalFastSa = C<opend0*0.995;	
	case 51:		PatternDirectionalFastSa = C>opend0*1.01;
	case -51:		PatternDirectionalFastSa = C<opend0*0.99;

	case 52:		PatternDirectionalFastSa = true;
	case -52:		PatternDirectionalFastSa = true;
	case >52:		PatternDirectionalFastSa = false;
	case <-52:		PatternDirectionalFastSa = false;
end;
