var: IsStartOfSession(true);
array: ohlcValues[23](0);

inputs: sessionStartTimeC(1700),sessionEndTimeC(1600);	// has to be set correctly to trading session times !
IsStartOfSession = _OHLCMulti5(sessionStartTimeC, sessionEndTimeC, ohlcValues);

//Works using 30-minute bars

inputs:
	MyContracts(1),
	MyPtnLY(152), MyPtnSY(100), MyPtnLN(8), MyPtnSN(109),
	MyStartTime(1800), MyEndTime(400),
	DaytoFilter(-1),
	MyStop(3000), MyProfit(6000),
	Length(20), NumDevs(2);

variables:
UpperBand(0),LowerBand(0); 

UpperBand = BollingerBand(c, Length, +NumDevs) ;
LowerBand = BollingerBand(c, Length, -NumDevs) ;

//Being an unmirrorerd script we use patterns why are independent on long and short
if dayofweek(date) <> DaytoFilter and tw(MyStartTime,MyEndTime) then begin
	if C crosses under UpperBand and PatternFast(MyPtnSY, ohlcValues) and PatternFast(MyPtnSN, ohlcValues)=False then Sell MyContracts contracts next bar market;
	if C crosses over LowerBand and PatternFast(MyPtnLY, ohlcValues) and PatternFast(MyPtnLN, ohlcValues)=False then Buy MyContracts contracts next bar market;
end;

setstopcontract;
if MyStop > 0 then setstoploss(MyStop);
If MyProfit > 0 then setprofittarget(MyProfit);

//TITAN FAST EXPORT (mette il report direttamente in C:\Titan\Reports)
input: TitanExportMode(0);
if TitanExportMode=1 then WriteDailiesCTitanReports(1100101, TitanExportMode);
