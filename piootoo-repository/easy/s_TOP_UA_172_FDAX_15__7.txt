//DEFINITIONS
MyDonchianUpper = HighestFC(high,DonchianPeriod);
MyDonchianLower = LowestFC(low,DonchianPeriod);

//CONDITIONS
if Time>=BeginTime and Time<EndTime then begin

  if sessionlastbar = false and PatternFast(MyPtnLY, ohlcValues) and PatternFast(MyPtnLN, ohlcValues)=False then 
  buy MySize contracts next bar MyDonchianUpper stop;
  
  if PatternFast(MyPtnSY, ohlcValues) and PatternFast(MyPtnSN, ohlcValues)=False then
  sellshort MySize contracts next bar MyDonchianLower stop;
end;

if marketposition = -1 and openpositionprofit > PP and PP>0 and PatterndirectionalFast(MyPtnSx, ohlcValues) then buy to cover next bar at market;
