//set 5 min chart exchange
input: MyContracts(1), StopLoss(7.5), BreakEven(15), TakeProfit(45), MonthUno(7), MonthDue(8);
vars: okl(true), oks(true), MyDOW(0);

if isStartOfSession[1] then begin 
	okl = true;
	oks = true;
	MyDOW = dayofweek(d)+1;
end;

if month(d)<>MonthUno and month(d)<>MonthDue and (Time<1530 or Time>=2100) 
and not (((closed1<(closed2-closed2*0.5/100))) or ((closed1>(closed2+closed2*0.5/100)))) then begin
	if ((highd0-opend0)>((highd1-opend1)*0.75)) and okl and {dayofweek(d)}MyDOW<>1 then buy("LE") MyContracts contracts next bar at highd1 stop;
	if ((opend0-lowd0)>((opend1-lowd1)*0.75)) and oks and {dayofweek(d)}MyDOW<>4 then sellshort("SE") MyContracts contracts next bar at lowd1 stop;
end;

if marketposition = 1 then begin
	okl = false;
	sell("LX stop") next bar (entryprice-StopLoss) stop;
end;

if marketposition = -1 then begin
	oks = false;
	buytocover("SX stop") next bar (entryprice+StopLoss) stop;
end;

setstopcontract;
setstoploss(StopLoss*1.2*bigpointvalue);
setbreakeven(BreakEven*bigpointvalue);
setprofittarget(TakeProfit*bigpointvalue);

if (time>=1654 and Time<1800) or isStartOfSession then begin    /// to avoid miss close in festivities
	if marketposition = 1 then sell ("LX") next bar at open;
	if marketposition = -1 then buytocover ("SX") next bar at open;
end;
