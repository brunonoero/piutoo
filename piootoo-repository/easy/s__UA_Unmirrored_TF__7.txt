vars: highd1 (0), lowd1 (0), IsStartOfSession(true);
array: ohlcValues[23](0);

//Works using 5 or 15 minute bars
Input: Mycontracts(1);
input: sessionStartTimeC(1700),sessionEndTimeC(1659);
Input: MyPtnLY(152), MyPtnSY(152), MyPtnLN(153), MyPtnSN(153);
Input: MyStartTrade(0000), MyEndTrade(2400), MyStartPause(1200), MyEndPause(1100), ID(0);
Input: MyStop(0), MyProfit(0);

IsStartOfSession = _OHLCMulti5(sessionStartTimeC, sessionEndTimeC, ohlcValues);
highd1 = ohlcvalues[5]; lowd1 = ohlcvalues[6]; //these are picked depending on the chosen trigger, in this example high and low of last session

//here the patterns are picked for each direction in an independent way so that the whole setup can be unmirrored
//in fact the general function Patternfast is used
If tw(MyStartTrade,MyEndTrade) and (Time < MyStartPause or time > MyEndPause) then begin
	If PatternFast(MyPtnLY, ohlcValues) and PatternFast(MyPtnLN, ohlcValues)=False then buy MyContracts contracts next bar at highd1 stop;
	If PatternFast(MyPtnSY, ohlcValues) and PatternFast(MyPtnSN, ohlcValues)=False then sellshort MyContracts contracts next bar at lowd1 stop;
end;

Setstopcontract;
If MyStop > 0 then setstoploss(MyStop);
If MyProfit > 0 then setprofittarget(MyProfit);

if ID=0 then setexitonclose;
