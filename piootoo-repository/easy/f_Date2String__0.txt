inputs: DateSelect( numericsimple ) ;                                
variables: var0( "" ), var1( "" ), var2( "" ) ;

var0 = NumToStr( 1900 + IntPortion( DateSelect * .0001 ), 0 ) ;	// year

if DateSelect >= 1000000 then                       
	var1 = MidStr( NumToStr( DateSelect, 0 ), 4, 2 )
else                       
	var1 = MidStr( NumToStr( DateSelect, 0 ), 3, 2 ) ;		// month
	
var2 = RightStr( NumToStr( DateSelect, 0 ), 2 ) ;			// day

Date2String = var2 + "/" + var1 + "/" + var0 ;
